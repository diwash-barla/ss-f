API Reference

Complete developer guide for the Spark Studio Elite Proxy API. Integrations available for Python, Node.js, and cURL.

Authentication (Bearer Token)

All endpoints are strictly protected by a proxy layer. You must pass your custom SS_API_KEY in the Authorization header of every request.

Authorization: Bearer <YOUR_SS_API_KEY>

* The API base URL is auto-detected below based on your current host.

POST /api/fast-search

Description

Performs a quick search across the database and returns matched URLs along with context snippets.

Parameters (JSON Body)

FieldTypeDescription
queryStringThe search keyword or question.
thresholdFloatRelevance threshold (e.g., 0.7). Optional.
curl -X POST "/api/fast-search" \ -H "Content-Type: application/json" \ -H "Authorization: Bearer YOUR_SS_API_KEY" \ -d '{ "query": "What is quantum mechanics?", "threshold": 0.7 }'
import requests url = "/api/fast-search" headers = { "Authorization": "Bearer YOUR_SS_API_KEY", "Content-Type": "application/json" } payload = { "query": "What is quantum mechanics?", "threshold": 0.7 } response = requests.post(url, json=payload, headers=headers) print(response.json())
const url = "/api/fast-search"; async function search() { const response = await fetch(url, { method: "POST", headers: { "Authorization": "Bearer YOUR_SS_API_KEY", "Content-Type": "application/json" }, body: JSON.stringify({ query: "What is quantum mechanics?" }) }); const data = await response.json(); console.log(data); }
JSON Response
{ "results": [ { "url": "https://example.com/quantum", "title": "Quantum Mechanics Explained", "snippet": "Quantum mechanics is a fundamental theory..." } ] }
POST /api/research/start

Description

Starts an asynchronous deep scanning job on the backend workers. You will receive a task_id immediately which you must use to poll for the result.

Parameters (JSON Body)

FieldTypeDescription
queryStringThe deep research topic.
curl -X POST "/api/research/start" \ -H "Content-Type: application/json" \ -H "Authorization: Bearer YOUR_SS_API_KEY" \ -d '{"query": "The Eridanus Supervoid anomaly"}'
import requests url = "/api/research/start" headers = {"Authorization": "Bearer YOUR_API_KEY"} payload = {"query": "The Eridanus Supervoid anomaly"} res = requests.post(url, json=payload, headers=headers) print(res.json()) # Returns task_id
fetch("/api/research/start", { method: "POST", headers: { "Authorization": "Bearer YOUR_KEY", "Content-Type": "application/json" }, body: JSON.stringify({ query: "The Eridanus Supervoid" }) }).then(r => r.json()).then(console.log);
JSON Response
{ "status": "processing", "task_id": "8f3a9b1c-..." }
GET /api/research/status/{task_id}

Description

Poll this endpoint every 2-3 seconds using the task_id to get live backend terminal logs and eventually the final AI Synthesis JSON.

curl -X GET "/api/research/status/TASK_ID_HERE" \ -H "Authorization: Bearer YOUR_SS_API_KEY"
import requests, time task_id = "YOUR_TASK_ID" url = f"/api/research/status/{task_id}" headers = {"Authorization": "Bearer YOUR_KEY"} while True: res = requests.get(url, headers=headers).json() if res["status"] == "complete": print("Done!", res["result"]) break time.sleep(3)
const url = `/api/research/status/${taskId}`; const poll = setInterval(async () => { const res = await fetch(url, { headers: { "Authorization": "Bearer KEY" }}); const data = await res.json(); if (data.status === "complete") { clearInterval(poll); console.log("Final Data:", data.result); } }, 3000);
JSON Response (When Complete)
{ "status": "complete", "logs": [ { "msg": "Extracting...", "color": "green" } ], "result": { "urls": [...], "data": { "conflict_engine": {...}, "lore_engine": {...} } } }
POST /api/extract

Description

Bypasses web firewalls to extract clean article text, image URLs, and video iframes from any public URL.

curl -X POST "/api/extract" \ -H "Authorization: Bearer YOUR_SS_API_KEY" \ -H "Content-Type: application/json" \ -d '{"url": "https://en.wikipedia.org/wiki/Earth"}'
import requests url = "/api/extract" payload = {"url": "https://en.wikipedia.org/wiki/Earth"} headers = {"Authorization": "Bearer KEY"} print(requests.post(url, json=payload, headers=headers).json())
JSON Response
{ "text": "Earth is the third planet...", "images": ["https://...image.jpg"], "videos": [] }
POST /api/generate-script

Description

Pass the JSON object obtained from Deep Search, and this AI will write a highly engaging YouTube or Short script complete with visual cues.

import requests url = "/api/generate-script" payload = { "query": "The Supervoid", "research_data": { /* Full Deep Scan Result JSON */ } } headers = {"Authorization": "Bearer KEY"} res = requests.post(url, json=payload, headers=headers) print(res.json().get("script"))
JSON Response
{ "script": "[Visual: Dark space]\n\nDid you know there is a void..." }
POST /api/dispatch-to-creator

Description

Sends the final edited script text to the Vidspark engine to render a full video.

FieldTypeDescription
script_textStringThe written script.
topicStringVideo title/topic.
orientationString"vertical" or "horizontal"
import requests url = "/api/dispatch-to-creator" payload = { "script_text": "Hello guys...", "topic": "My Video", "orientation": "horizontal" } headers = {"Authorization": "Bearer KEY"} requests.post(url, json=payload, headers=headers)
fetch("/api/dispatch-to-creator", { method: "POST", headers: { "Authorization": "Bearer KEY", "Content-Type": "application/json" }, body: JSON.stringify({ script_text: "...", topic: "...", orientation: "vertical" }) });
JSON Response
{ "status": "success", "task_id": "vid_xyz", "message": "Script queued for rendering." }
